**Performance Metrics:**

**Implementation:**
```python
from sklearn.linear_model import LogisticRegression

class CandidateMatchingBaseline:
    def __init__(self):
        self.model = LogisticRegression(max_iter=1000, random_state=42)
        self.scaler = StandardScaler()
    
    def train(self, X, y):
        """Train baseline Logistic Regression model"""
        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)
        
        # Evaluate
        train_score = self.model.score(X_scaled, y)
        return {"accuracy": train_score}
    
    def predict(self, features_dict):
        """Generate prediction for candidate-job match"""
        features_array = self.prepare_features(features_dict)
        features_scaled = self.scaler.transform([features_array])
        
        prediction = self.model.predict(features_scaled)[0]
        probability = self.model.predict_proba(features_scaled)[0]
        
        return {
            "prediction": int(prediction),
            "confidence": float(max(probability)),
            "probability": float(probability[1])
        }
```

**Implementation:**
```python
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler

class CandidateMatchingRandomForest:
    def __init__(self, n_estimators=100):
        self.model = RandomForestClassifier(
            n_estimators=n_estimators,
            max_depth=10,
            min_samples_split=5,
            min_samples_leaf=2,
            random_state=42,
            class_weight='balanced',
            n_jobs=-1
        )
        self.scaler = StandardScaler()
        self.feature_names = None
    
    def train(self, X, y):
        """Train Random Forest model with cross-validation"""
        X_scaled = self.scaler.fit_transform(X)
        
        # Training
        self.model.fit(X_scaled, y)
        
        # Evaluation
        from sklearn.model_selection import cross_val_score
        cv_scores = cross_val_score(self.model, X_scaled, y, cv=5)
        
        train_accuracy = self.model.score(X_scaled, y)
        
        return {
            "accuracy": train_accuracy,
            "cv_mean": cv_scores.mean(),
            "cv_std": cv_scores.std(),
            "cv_scores": cv_scores.tolist()
        }
    
    def predict(self, features_dict):
        """Generate prediction with confidence score"""
        features_array = self.prepare_features(features_dict)
        features_scaled = self.scaler.transform([features_array])
        
        prediction = self.model.predict(features_scaled)[0]
        probability = self.model.predict_proba(features_scaled)[0]
        
        return {
            "prediction": int(prediction),
            "confidence": float(max(probability)),
            "probability": float(probability[1]),
            "match_score": float(probability[1]) * 100  # 0-100 scale
        }
    
    def get_feature_importance(self, top_n=10):
        """Extract top N most important features"""
        importances = self.model.feature_importances_
        indices = np.argsort(importances)[-top_n:][::-1]
        
        return {
            "features": [self.feature_names[i] for i in indices],
            "importance": [float(importances[i]) for i in indices]
        }
    
    def save(self, filepath):
        """Save trained model to disk"""
        import pickle
        with open(filepath, 'wb') as f:
            pickle.dump(self, f)
    
    @staticmethod
    def load(filepath):
        """Load previously trained model"""
        import pickle
        with open(filepath, 'rb') as f:
            return pickle.load(f)
```

**Hyperparameters:**
- `n_estimators`: 100 trees
- `max_depth`: 10 (prevent overfitting)
- `min_samples_split`: 5 (minimum samples to split)
- `min_samples_leaf`: 2 (minimum samples in leaf)
- `class_weight`: balanced (handle class imbalance)
- `random_state`: 42 (reproducibility)

def get_feature_importance(self, top_n=10):
    """Get top N most important features from Random Forest"""
    importances = self.model.feature_importances_
    feature_importance = {
        name: float(imp) 
        for name, imp in zip(self.feature_names, importances)

    # Sort and return top N
    sorted_features = sorted(
        feature_importance.items(), 
        key=lambda x: x[1],

```python
def prepare_data(self, training_data):
    """Extract features and labels from raw training data"""
    X = []
    y = []
    self.feature_names = []
    
    for sample in training_data:
        # Extract features
        features = [
            sample['years_experience'],
            sample['skill_count'],
            sample['soft_skill_count'],
            sample['has_certification'],
            sample['is_remote'],
            sample['technical_skill_score'],
            sample['soft_skill_score'],
            # ... individual skill indicators (30+ features)
        ]
        
        X.append(features)
        y.append(sample['label'])
    
    return np.array(X), np.array(y)
```

**Feature Matrix:**
- Rows: 500 samples (candidates)
- Columns: 45 features
- Target: Binary label (0=non-match, 1=match)

    X_train, X_test, y_train, y_test = train_test_split(

    )
    
    # Train baseline model (DEV-01)
    baseline = CandidateMatchingBaseline()
    baseline_metrics = baseline.train(X_train, y_train)
    baseline_test_acc = baseline.model.score(
        baseline.scaler.transform(X_test), y_test
    )
    
    # Train core model (DEV-02)
    core_model = CandidateMatchingRandomForest(n_estimators=100)
    core_metrics = core_model.train(X_train, y_train)

    return {
        "baseline": baseline_metrics,
        "core": core_metrics,
        "improvement": improvement,
        "baseline_model": baseline,
        "core_model": core_model
    }

    """Score multiple candidates for a job"""
    results = []
    
    for candidate in candidates_list:
        prediction = model.predict(candidate)
        results.append(prediction)
    
    # Sort by confidence
    results.sort(key=lambda x: x['confidence'], reverse=True)
    
    return results
